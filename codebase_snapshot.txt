// Receives roster, mailing-list, and community/volunteer submissions and stores them in Vercel KV
// (Upstash Redis) via the REST API. No dependencies, plain fetch.

const URL_ = process.env.KV_REST_API_URL;
const TOKEN = process.env.KV_REST_API_TOKEN;

async function kv(cmd) {
  const r = await fetch(URL_, {
    method: 'POST',
    headers: { Authorization: 'Bearer ' + TOKEN, 'Content-Type': 'application/json' },
    body: JSON.stringify(cmd),
  });
  return r.json();
}

const clip = (s, n) => (s || '').toString().trim().slice(0, n);

module.exports = async function handler(req, res) {
  if (req.method !== 'POST') { res.status(405).json({ error: 'method' }); return; }
  if (!URL_ || !TOKEN) { res.status(500).json({ error: 'store not configured' }); return; }

  let b = req.body;
  if (!b || typeof b === 'string') { try { b = JSON.parse(b || '{}'); } catch (e) { b = {}; } }

  const email = clip(b.email, 200);
  if (!email || email.indexOf('@') < 0) { res.status(400).json({ error: 'email required' }); return; }

  let type = 'roster';
  if (b.type === 'mailing') type = 'mailing';
  else if (b.type === 'community' || b.type === 'contact') type = 'community';

  let rec;
  if (type === 'mailing') {
    rec = { ts: Date.now(), email };
  } else if (type === 'community') {
    rec = {
      ts: Date.now(),
      name: clip(b.name, 200),
      email,
      interest: clip(b.subject || b.interest, 200),
      message: clip(b.message, 3000)
    };
  } else {
    rec = {
      ts: Date.now(),
      name: clip(b.name, 200),
      email,
      act: clip(b.act, 200),
      discipline: clip(b.discipline, 200),
      links: clip(b.links, 500),
      notes: clip(b.notes, 2000)
    };
  }

  try {
    await kv(['RPUSH', 'smnw:' + type, JSON.stringify(rec)]);
    res.status(200).json({ success: true });
  } catch (e) {
    res.status(500).json({ error: 'save failed' });
  }
};

function parseMultipartOrJson(req) {
  return new Promise((resolve) => {
    let raw = '';
    req.on('data', chunk => raw += chunk);
    req.on('end', () => {
      const contentType = req.headers['content-type'] || '';
      const fields = {};

      if (contentType.includes('application/json')) {
        try { Object.assign(fields, JSON.parse(raw)); } catch(e) {}
      } else if (contentType.includes('multipart/form-data')) {
        const boundaryMatch = contentType.match(/boundary=(?:"([^"]+)"|([^;]+))/i);
        const boundary = boundaryMatch ? (boundaryMatch[1] || boundaryMatch[2]) : null;
        if (boundary) {
          const parts = raw.split('--' + boundary);
          for (const part of parts) {
            const nameMatch = part.match(/name="([^"]+)"/);
            if (nameMatch) {
              const fieldName = nameMatch[1];
              const bodyStart = part.indexOf('\r\n\r\n');
              if (bodyStart !== -1) {
                // If this part is a file, skip body text extraction here
                if (!part.includes('filename=')) {
                  const val = part.slice(bodyStart + 4).replace(/\r\n$/, '').trim();
                  fields[fieldName] = val;
                }
              }
            }
          }
        }
      } else if (req.body && typeof req.body === 'object') {
        Object.assign(fields, req.body);
      }
      resolve(fields);
    });
  });
}

module.exports = async function handler(req, res) {
  res.setHeader('Access-Control-Allow-Origin', '*');
  res.setHeader('Access-Control-Allow-Methods', 'POST, OPTIONS');
  res.setHeader('Access-Control-Allow-Headers', 'Content-Type');

  if (req.method === 'OPTIONS') return res.status(200).end();
  if (req.method !== 'POST') return res.status(405).json({ success: false, message: 'Method not allowed' });

  try {
    const campaignData = await parseMultipartOrJson(req);

    const championName = campaignData.champion_name || campaignData.name || 'Community Champion';
    const championEmail = campaignData.champion_email || campaignData.email || 'info@sustainablemusicnw.org';
    const neighborhoodTitle = campaignData.neighborhood_title || campaignData.neighborhood || 'Block Party';
    const targetDate = campaignData.target_event_date || campaignData.date || 'Upcoming 2026';
    const customDesc = campaignData.description || 
      `Funding live, fair-wage music on our street in ${neighborhoodTitle} with Sustainable Music Northwest!`;

    const campaignId = `cmp_${Date.now()}`;
    const pdfFileName = `Flyer-${campaignId}.pdf`;
    const pdfFilePath = path.join(storageDir, pdfFileName);

    let givebutterUrl = `https://givebutter.com/blockstage-${campaignId}`;
    let givebutterCode = `gb_${campaignId}`;

    // 1. Provision Givebutter campaign
    const apiKey = process.env.GIVEBUTTER_API_KEY ? process.env.GIVEBUTTER_API_KEY.trim() : null;
    if (apiKey) {
      try {
        const gbPayload = {
          title: `${neighborhoodTitle} Block Stage Concert`,
          description: customDesc,
          goal: 1500,
          type: 'fundraise'
        };

        const gbResponse = await axios.post(
          'https://api.givebutter.com/v1/campaigns',
          gbPayload,
          {
            headers: {
              'Authorization': `Bearer ${apiKey}`,
              'Content-Type': 'application/json'
            },
            timeout: 9000
          }
        );

        givebutterUrl = gbResponse.data?.data?.url || gbResponse.data?.url || gbResponse.data?.link || givebutterUrl;
        givebutterCode = gbResponse.data?.data?.code || gbResponse.data?.code || gbResponse.data?.id || givebutterCode;
      } catch (gbErr) {
        console.warn('Givebutter provisioning warning (using fallback):', gbErr.response ? gbErr.response.data : gbErr.message);
      }
    }

    // 2. Render dynamic PDF flyer via flyerEngine
    let flyerDataUri = null;
    try {
      const pdfBuffer = await generateBlockStageFlyer({
        champion_name: championName,
        champion_email: championEmail,
        neighborhood_title: neighborhoodTitle,
        target_event_date: targetDate,
        description: customDesc,
        givebutter_url: givebutterUrl
      }, pdfFilePath);

      // Encode buffer directly as a data URI so downloads work in serverless without disk persistence
      const base64Pdf = Buffer.isBuffer(pdfBuffer) 
        ? pdfBuffer.toString('base64') 
        : (fs.existsSync(pdfFilePath) ? fs.readFileSync(pdfFilePath).toString('base64') : null);

      if (base64Pdf) {
        flyerDataUri = `data:application/pdf;base64,${base64Pdf}`;
      }
    } catch (flyerErr) {
      console.error('Flyer generation warning:', flyerErr.message);
    }

    // 3. Dispatch champion welcome email via emailEngine
    if (process.env.EMAIL_USER && process.env.EMAIL_PASS) {
      try {
        await sendChampionWelcomeEmail({
          champion_name: championName,
          champion_email: championEmail,
          neighborhood_title: neighborhoodTitle,
          target_event_date: targetDate,
          givebutter_url: givebutterUrl
        }, pdfFilePath);
        console.log(`Welcome email successfully dispatched to ${championEmail}`);
      } catch (emailErr) {
        console.error('SMTP Email Error (non-blocking):', emailErr.message);
      }
    }

    // 4. Return success response to browser
    return res.status(200).json({
      success: true,
      campaign_id: campaignId,
      givebutter_code: givebutterCode,
      givebutter_url: givebutterUrl,
      flyer_pdf_url: flyerDataUri || `/storage/${pdfFileName}`,
      message: 'Campaign provisioned and flyer generated successfully.'
    });

  } catch (err) {
    console.error('Error creating campaign:', err);
    return res.status(500).json({ 
      success: false, 
      message: err.message || 'Failed to provision campaign and render flyer.' 
    });
  }
};

const PDFDocument = require('pdfkit');
const QRCode = require('qrcode');
const fs = require('fs');
const path = require('path');

/**
 * Dynamically measures rendered string height inside PDFKit to guarantee headline fit.
 */
function calculateDynamicFontSize(doc, text, maxWidth, maxHeight) {
    let fontSize = 28;
    while (fontSize > 12) {
        doc.fontSize(fontSize).font('Helvetica-Bold');
        const height = doc.heightOfString(text, { width: maxWidth, lineGap: -1 });
        if (height <= maxHeight) break;
        fontSize -= 1;
    }
    let yPos = 188;
    if (fontSize < 16) yPos = 182;
    else if (fontSize < 22) yPos = 184;
    else if (fontSize < 26) yPos = 186;
    return { fontSize, yPos };
}

/**
 * Generates an 8.5" x 11" flyer returning a Buffer with full 7-tier alignment.
 */
async function generateBlockStageFlyer(campaignData, outputPath = null) {
    return new Promise(async (resolve, reject) => {
        try {
            const doc = new PDFDocument({ size: 'LETTER', margin: 0 });
            const buffers = [];

            doc.on('data', buffers.push.bind(buffers));
            doc.on('end', () => {
                const pdfBuffer = Buffer.concat(buffers);
                if (outputPath) {
                    try {
                        fs.writeFileSync(outputPath, pdfBuffer);
                    } catch (e) {
                        console.warn('Could not write flyer to filesystem path:', e.message);
                    }
                }
                resolve(pdfBuffer);
            });

            // 1. Render Base Background Graphic
            const possiblePaths = [
                path.join(__dirname, 'templates', 'New Template Block Stage.jpg'),
                path.join(__dirname, 'New Template Block Stage.jpg'),
                path.join(process.cwd(), 'templates', 'New Template Block Stage.jpg'),
                path.join(process.cwd(), 'New Template Block Stage.jpg')
            ];

            const templatePath = possiblePaths.find(p => fs.existsSync(p));

            if (templatePath) {
                doc.image(templatePath, 0, 0, { width: 612, height: 792 });
            } else {
                // Layout Canvas fallback if image is not located
                doc.rect(0, 0, 612, 792).fill('#F8F7F4');
                doc.fillColor('#18181B').font('Helvetica-Bold').fontSize(24).text('BLOCK STAGE CONCERT', 50, 40);
            }

            // 2. Headline Overlay
            const rawTitle = campaignData.neighborhood_title ? campaignData.neighborhood_title.trim() : 'OUR BLOCK';
            const headlineText = `${rawTitle.toUpperCase()}!`;
            const { fontSize, yPos } = calculateDynamicFontSize(doc, headlineText, 521.28, 52);

            doc.fillColor('#182723')
               .font('Helvetica-Bold')
               .fontSize(fontSize)
               .text(headlineText, 43.2, yPos, {
                   width: 521.28,
                   height: 52,
                   align: 'center',
                   lineGap: -1,
                   ellipsis: true
               });

            // 3. Cutoff Date Overlay
            const cutoffDate = campaignData.target_event_date || 'Upcoming 2026';
            doc.fillColor('#182723')
               .font('Helvetica-Bold')
               .fontSize(18)
               .text(cutoffDate, 233.5, 245, {
                   width: 194.9,
                   align: 'left'
               });

            // 4. Host Name Overlay
            const champName = campaignData.champion_name || 'Community Host';
            doc.fillColor('#E85E2F')
               .font('Helvetica')
               .fontSize(15)
               .text(champName, 44.4, 700, {
                   width: 191.3,
                   align: 'center'
               });

            // 5. Host Email Overlay
            const champEmail = campaignData.champion_email || 'info@sustainablemusicnw.org';
            doc.fillColor('#E85E2F')
               .font('Helvetica')
               .fontSize(15)
               .text(champEmail, 256.1, 700, {
                   width: 191.3,
                   align: 'center'
               });

            // 6. Dynamic QR Code Overlay
            const pledgeUrl = campaignData.givebutter_url || 'https://givebutter.com';
            const qrBuffer = await QRCode.toBuffer(pledgeUrl, {
                width: 200,
                margin: 1,
                color: {
                    dark: '#182723',
                    light: '#FFFFFF'
                }
            });

            doc.image(qrBuffer, 468.1, 632.7, { width: 104.4, height: 104.4 });

            doc.end();
        } catch (err) {
            reject(err);
        }
    });
}
